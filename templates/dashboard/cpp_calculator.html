Calculator using C++ GUI (Qt)

This example builds a desktop calculator using Qt Widgets in C++. It supports basic operations: addition, subtraction, multiplication, and division.

  1. Install Qt Creator and Qt 6 (or Qt 5).
  2. Create a new Qt Widgets Application.
  3. Replace your main.cpp with the code below.
  4. Build and run the project.
main.cpp
#include <QApplication>
#include <QWidget>
#include <QGridLayout>
#include <QPushButton>
#include <QLineEdit>
#include <QString>

class Calculator : public QWidget {
public:
    Calculator() {
        auto *layout = new QGridLayout(this);
        display = new QLineEdit(this);
        display->setReadOnly(true);
        display->setAlignment(Qt::AlignRight);
        display->setMinimumHeight(40);

        layout->addWidget(display, 0, 0, 1, 4);

        const QString buttons[4][4] = {
            {"7", "8", "9", "/"},
            {"4", "5", "6", "*"},
            {"1", "2", "3", "-"},
            {"0", "C", "=", "+"}
        };

        for (int row = 0; row < 4; ++row) {
            for (int col = 0; col < 4; ++col) {
                auto *btn = new QPushButton(buttons[row][col], this);
                btn->setMinimumSize(60, 45);
                layout->addWidget(btn, row + 1, col);
                connect(btn, &QPushButton::clicked, this, [this, btn]() {
                    handleButton(btn->text());
                });
            }
        }

        setWindowTitle("C++ GUI Calculator");
        resize(300, 260);
    }

private:
    QLineEdit *display;
    double firstValue = 0.0;
    QString operation;
    bool waitingSecondValue = false;

    void handleButton(const QString &text) {
        if (text == "C") {
            display->clear();
            firstValue = 0.0;
            operation.clear();
            waitingSecondValue = false;
            return;
        }

        if (text == "+" || text == "-" || text == "*" || text == "/") {
            firstValue = display->text().toDouble();
            operation = text;
            waitingSecondValue = true;
            return;
        }

        if (text == "=") {
            double secondValue = display->text().toDouble();
            double result = 0.0;

            if (operation == "+") result = firstValue + secondValue;
            if (operation == "-") result = firstValue - secondValue;
            if (operation == "*") result = firstValue * secondValue;
            if (operation == "/") {
                if (secondValue == 0.0) {
                    display->setText("Error");
                    return;
                }
                result = firstValue / secondValue;
            }

            display->setText(QString::number(result));
            waitingSecondValue = false;
            return;
        }

        if (waitingSecondValue) {
            display->clear();
            waitingSecondValue = false;
        }
        display->setText(display->text() + text);
    }
};

int main(int argc, char *argv[]) {
    QApplication app(argc, argv);
    Calculator window;
    window.show();
    return app.exec();
}
